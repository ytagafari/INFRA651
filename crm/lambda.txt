def get_account_costs(ce_client, start_cur, end_cur, start_prev, end_prev, start_full_prev, end_full_prev, current_day, days_in_month):
    metrics = {
        'total_mtd': 0.0, 'total_prev_same_day': 0.0, 'total_prev_full': 0.0, 'service_rows': [], 'three_month_history': []
    }
    
    prev_same_day_map = {}
    prev_full_map = {}
    current_services = {}

    try:
        # 1. Current Month-to-Date
        r1 = ce_client.get_cost_and_usage(
            TimePeriod={'Start': start_cur, 'End': end_cur}, 
            Granularity='MONTHLY', Metrics=['UnblendedCost'],
            GroupBy=[{'Type': 'DIMENSION', 'Key': 'SERVICE'}]
        )
        for group in r1['ResultsByTime'][0]['Groups']:
            svc_name = group['Keys'][0]
            amt = float(group['Metrics']['UnblendedCost']['Amount'])
            current_services[svc_name] = amt
            metrics['total_mtd'] += amt

        # 2. Previous Month Same Period
        r2 = ce_client.get_cost_and_usage(
            TimePeriod={'Start': start_prev, 'End': end_prev}, 
            Granularity='MONTHLY', Metrics=['UnblendedCost'],
            GroupBy=[{'Type': 'DIMENSION', 'Key': 'SERVICE'}]
        )
        for group in r2['ResultsByTime'][0]['Groups']:
            svc_name = group['Keys'][0]
            amt = float(group['Metrics']['UnblendedCost']['Amount'])
            prev_same_day_map[svc_name] = amt
            metrics['total_prev_same_day'] += amt

        # 3. Full Closed Last Month
        r3 = ce_client.get_cost_and_usage(
            TimePeriod={'Start': start_full_prev, 'End': end_full_prev}, 
            Granularity='MONTHLY', Metrics=['UnblendedCost'],
            GroupBy=[{'Type': 'DIMENSION', 'Key': 'SERVICE'}]
        )
        for group in r3['ResultsByTime'][0]['Groups']:
            svc_name = group['Keys'][0]
            amt = float(group['Metrics']['UnblendedCost']['Amount'])
            prev_full_map[svc_name] = amt
            metrics['total_prev_full'] += amt

        # 4. Historical 3-Month Trend Lookup (For Monthly Report)
        history_start = (datetime.strptime(start_full_prev, '%Y-%m-%d') - timedelta(days=90)).strftime('%Y-%m-01')
        r_hist = ce_client.get_cost_and_usage(
            TimePeriod={'Start': history_start, 'End': start_cur},
            Granularity='MONTHLY', Metrics=['UnblendedCost']
        )
        for period in r_hist['ResultsByTime']:
            m_label = datetime.strptime(period['TimePeriod']['Start'], '%Y-%m-%d').strftime('%b')
            m_amt = float(period['Total']['UnblendedCost']['Amount'])
            metrics['three_month_history'].append((m_label, m_amt))

        # 5. Build Row Aggregations
        all_services = set(current_services.keys()) | set(prev_same_day_map.keys()) | set(prev_full_map.keys())

        for svc in all_services:
            mtd = current_services.get(svc, 0.0)
            prev_same_day = prev_same_day_map.get(svc, 0.0)
            prev_full = prev_full_map.get(svc, 0.0)
            
            status = "Active" if mtd > 0.0 else "Inactive"
            prediction = (mtd / current_day) * days_in_month if mtd > 0 else 0.0
            
            # Missing Item 1: Amount Delta + Percentage Delta
            delta_amt = mtd - prev_same_day
            delta_pct = ((mtd - prev_same_day) / prev_same_day) * 100 if prev_same_day > 0 else (100.0 if mtd > 0 else 0.0)
            
            if mtd == 0 and prev_same_day == 0 and prev_full == 0:
                continue

            metrics['service_rows'].append({
                'name': svc, 'status': status, 'mtd': mtd, 'prev_same_day': prev_same_day,
                'delta_amt': delta_amt, 'delta_pct': delta_pct, 'prediction': prediction, 'prev_full': prev_full
            })

        metrics['service_rows'].sort(key=lambda x: x['mtd'], reverse=True)
        metrics['service_rows'] = metrics['service_rows'][:12]

    except Exception as e:
        print(f"Error components tracking: {str(e)}")
        
    return metrics

def generate_html_table(service_rows):

        return "<p style='padding-left:15px; color:#555;'>No data compiled.</p>"
        
    html = """

        <thead>

                <th style="padding: 10px; text-align: left; font-weight: bold; border-right: 1px solid #a0a0a0; color: #333;">Status</th>
                <th style="padding: 10px; text-align: left; font-weight: bold; border-right: 1px solid #a0a0a0; color: #333;">Service Infrastructure</th>
                <th style="padding: 10px; text-align: right; border-right: 1px solid #a0a0a0; color: #333;">Cost MTD</th>
                <th style="padding: 10px; text-align: right; border-right: 1px solid #a0a0a0; color: #555;">LM Same Day</th>
                <th style="padding: 10px; text-align: right; border-right: 1px solid #a0a0a0; color: #b33939;">MoM Delta (Amt / %)</th>
                <th style="padding: 10px; text-align: right; border-right: 1px solid #a0a0a0; color: #0073bb;">EOM Predict</th>
                <th style="padding: 10px; text-align: right; color: #333;">Total Last Month</th>
            </tr>
        </thead>
        <tbody>
    """
    for r in service_rows:
        status_color = "#2ed573" if r['status'] == "Active" else "#747d8c"
        delta_color = "#2ed573" if r['delta_amt'] < 0 else ("#ff4757" if r['delta_amt'] > 0 else "#333")
        
        # Show both Amount and Percentage in Delta column
        delta_str = f"${r['delta_amt']:+,.2f} ({r['delta_pct']:+.2f}%)" if r['prev_same_day'] > 0 or r['mtd'] > 0 else "$0.00 (0.00%)"
        
        html += f"""
            <tr style="border-bottom: 1px solid #d0d0d0;">
                <td style="padding: 8px 10px; text-align: left; border-right: 1px solid #a0a0a0; font-weight: bold; color: {status_color}; white-space: nowrap;">● {r['status']}</td>
                <td style="padding: 8px 10px; text-align: left; border-right: 1px solid #a0a0a0; color: #333; font-weight: 500;">{r['name']}</td>
                <td style="padding: 8px 10px; text-align: right; border-right: 1px solid #a0a0a0; color: #333; font-family: monospace;">${r['mtd']:,.2f}</td>
                <td style="padding: 8px 10px; text-align: right; border-right: 1px solid #a0a0a0; color: #666; font-family: monospace;">${r['prev_same_day']:,.2f}</td>
                <td style="padding: 8px 10px; text-align: right; border-right: 1px solid #a0a0a0; color: {delta_color}; font-family: monospace; font-weight: bold;">{delta_str}</td>
                <td style="padding: 8px 10px; text-align: right; border-right: 1px solid #a0a0a0; color: #0073bb; font-family: monospace; font-weight: bold;">${r['prediction']:,.2f}</td>
                <td style="padding: 8px 10px; text-align: right; color: #333; font-family: monospace;">${r['prev_full']:,.2f}</td>
            </tr>
        """
    html += "</tbody></table>"
    return html

def make_ascii_chart(history_data):
    """Missing Item 3: Renders a clean visual trend bar summary inside HTML using inline CSS tags"""
    if not history_data: return ""
    chart_html = "<div style='font-family:monospace; font-size:13px; line-height:1.8; color:#333; margin-top:10px;'>"
    max_val = max([val for _, val in history_data]) if history_data else 1
    for month, val in history_data:
        bar_length = int((val / max_val) * 20) if max_val > 0 else 0
        bar = "■" * bar_length
        chart_html += f"&nbsp;&nbsp;{month:<4} | {bar:<20} ${val:,.2f}<br>"
    chart_html += "</div>"
    return chart_html
